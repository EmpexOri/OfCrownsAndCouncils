je_expand_secular_admin = {
    name = "je_expand_secular_admin"
    desc = "je_expand_secular_admin_desc"

    visible = {
        always = yes
    }

    possible = {
        NOT = { has_modifier = sax_secular_administration_expanded }
        NOT = { 
            OR = {
                has_law_or_variant = law_type:law_state_religion 
                has_law_or_variant = law_type:law_freedom_of_conscience 
            }
        }
        NOT = {
            has_law_or_variant = law_type:law_hereditary_bureaucrats
        }
    }

    effect = {
        add_modifier = {
            name = sax_secular_administration_expanded
            years = 10
        }
    }
}
